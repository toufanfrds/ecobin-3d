// ====================================================================
// PROYEK TEKNIK: ECOBIN - TEMPAT SAMPAH PINTAR GALON BEKAS
// Komponen:
// - Sensor Sentuh / Proksimitas Depan (Digital Pin 2)
// - Saklar Pemilih Kategori Sampah    (Digital Pin 4)
// - Motor Servo Pembuka Tutup         (PWM Pin 9)
// - Lampu Indikator LED Hijau Depan   (Digital Pin 13)
// - Layar LCD Karakter 16x2 I2C       (Alamat 0x27, SDA A4, SCL A5)
// ====================================================================

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Servo.h>

// Inisialisasi LCD 16x2 I2C pada alamat 0x27
LiquidCrystal_I2C lcd(0x27, 16, 2);
Servo servoTutup;

// Definisi Pin Mikrokontroler Arduino Uno / Nano
const int PIN_SENSOR    = 2;   // Sensor Sentuh / Proksimitas Depan
const int PIN_SW_JENIS  = 4;   // Saklar Pilih Jenis (HIGH = Organik, LOW = Anorganik)
const int PIN_SERVO     = 9;   // Motor Servo Penggerak Tuas Tutup
const int PIN_LED_HIJAU = 13;  // Lampu Indikator LED Hijau di Depan Galon

// Konstanta Sudut Buka & Tutup Servo
const int SUDUT_TUTUP = 0;
const int SUDUT_BUKA  = 80;

void setup() {
  pinMode(PIN_SENSOR, INPUT_PULLUP);
  pinMode(PIN_SW_JENIS, INPUT_PULLUP);
  pinMode(PIN_LED_HIJAU, OUTPUT);

  // Posisi awal tertutup & LED mati
  digitalWrite(PIN_LED_HIJAU, LOW);
  servoTutup.attach(PIN_SERVO);
  servoTutup.write(SUDUT_TUTUP);

  // Inisialisasi tampilan awal LCD 16x2
  lcd.init();
  lcd.noBacklight();
  tampilStandby();
}

void loop() {
  // Baca status sensor depan (aktif LOW saat ditekan/tersentuh)
  bool sensorAktif = (digitalRead(PIN_SENSOR) == LOW);
  bool modeOrganik = (digitalRead(PIN_SW_JENIS) == HIGH);

  if (sensorAktif) {
    // 1. Nyalakan Lampu LED Hijau Depan & Backlight LCD
    digitalWrite(PIN_LED_HIJAU, HIGH);
    lcd.backlight();

    // 2. Tampilkan status terbuka dan kategori sampah pada LCD 16x2
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("STATUS: TERBUKA ");
    lcd.setCursor(0, 1);
    if (modeOrganik) {
      lcd.print("JENIS: ORGANIK  ");
    } else {
      lcd.print("JENIS: ANORGANIK");
    }

    // 3. Gerakkan tuas motor servo untuk membuka tutup galon (80 derajat)
    servoTutup.write(SUDUT_BUKA);
    delay(4000); // Tahan tutup terbuka selama 4 detik

    // 4. Tutup kembali galon, matikan LED hijau & kembalikan LCD ke standby
    servoTutup.write(SUDUT_TUTUP);
    digitalWrite(PIN_LED_HIJAU, LOW);
    tampilStandby();
    lcd.noBacklight();
  }

  delay(50);
}

void tampilStandby() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("ECOBIN READY    ");
  lcd.setCursor(0, 1);
  lcd.print("TEKAN SENSOR    ");
}
